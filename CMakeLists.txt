cmake_minimum_required(VERSION 3.20)
project(exchange_sandbox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SANDBOX_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(sandbox INTERFACE)
target_include_directories(sandbox INTERFACE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(sandbox INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(SANDBOX_SANITIZERS)
    target_compile_options(sandbox INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sandbox INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_executable(simulate src/simulate.cpp)
target_link_libraries(simulate PRIVATE sandbox)
add_executable(sandbox_tests tests/tests.cpp)
target_link_libraries(sandbox_tests PRIVATE sandbox)
enable_testing()
add_test(NAME sandbox_tests COMMAND sandbox_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME simulation_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:simulate>)
endif()
